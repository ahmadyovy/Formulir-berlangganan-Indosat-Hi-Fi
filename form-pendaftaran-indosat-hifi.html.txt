    <script>
        let currentWauUrl = "";
        let rawFormattedText = "";

        function sendToWhatsApp(event) {
            event.preventDefault();

            // Ambil data dari form
            const nama = document.getElementById('nama').value.trim();
            const nowa = document.getElementById('nowa').value.trim();
            const email = document.getElementById('email').value.trim();
            const alamat = document.getElementById('alamat').value.trim();
            
            // Ambil paket yang dipilih
            const selectedPaket = document.querySelector('input[name="paket"]:checked').value;

            if (!nama || !nowa || !email || !alamat) {
                alert("Mohon lengkapi semua bidang data diri terlebih dahulu.");
                return;
            }

            // Susun format pesan rapi ke WhatsApp 082258659397
            const textLines = [
                "✨ *PENDAFTARAN INDOSAT HI-FI* ✨",
                "",
                "PAKET YANG DI PILIH :",
                selectedPaket,
                "",
                "Nama : " + nama,
                "Alamat : " + alamat,
                "Email : " + email,
                "No. Tlp/wa : " + nowa,
                "",
                "Klik kirim via whatsapp (082258659397) Yovy"
            ];

            rawFormattedText = textLines.join("\n");
            const targetPhone = "6282258659397";
            
            // Buat URL WhatsApp
            currentWauUrl = "https://api.whatsapp.com/send?phone=" + targetPhone + "&text=" + encodeURIComponent(rawFormattedText);

            // Tampilkan modal backup jika browser membatasi direct redirect
            document.getElementById('modalTextarea').value = rawFormattedText;
            document.getElementById('copyModal').classList.remove('hidden');

            // Coba buka WhatsApp otomatis
            window.open(currentWauUrl, '_blank');
        }

        function closeModal() {
            document.getElementById('copyModal').classList.add('hidden');
            document.getElementById('copyStatus').classList.add('hidden');
        }

        function redirectToWA() {
            if (currentWauUrl) {
                window.location.href = currentWauUrl;
            }
        }

        function copyToClipboard() {
            const textarea = document.getElementById('modalTextarea');
            textarea.select();
            textarea.setSelectionRange(0, 99999);
            
            try {
                document.execCommand('copy');
                const status = document.getElementById('copyStatus');
                status.classList.remove('hidden');
                setTimeout(() => status.classList.add('hidden'), 3000);
            } catch (err) {
                console.error('Gagal menyalin:', err);
            }
        }
    </script>
